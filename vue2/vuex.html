Vuex 核心源码实现

本示例展示 Vuex 状态管理的核心实现原理，包含 State、Getters、Mutations、Actions 等核心概念， 以及响应式状态更新机制。右侧演示区域展示了一个完整的计数器应用。

Vuex 核心实现代码

// 简化版 Vuex 实现
class Store {
  constructor(options = {}) {
    // 初始化 state (使用 Vue 的响应式系统)
    this._vm = new Vue({
      data: {
        $$state: options.state
      }
    });

    // 保存 mutations 和 actions
    this._mutations = options.mutations || {};
    this._actions = options.actions || {};
    
    // 绑定 commit 和 dispatch 的上下文
    this.commit = this.commit.bind(this);
    this.dispatch = this.dispatch.bind(this);
    
    // 初始化 getters
    this.getters = {};
    const computed = {};
    const store = this;
    
    if (options.getters) {
      Object.keys(options.getters).forEach(key => {
        // 使用计算属性实现 getters 缓存
        computed[key] = function() {
          return options.getters[key](store.state);
        };
        
        // 为 getters 定义只读属性
        Object.defineProperty(store.getters, key, {
          get: () => store._vm[key],
          enumerable: true
        });
      });
    }
    
    // 将 getters 挂载到 Vue 实例
    this._vm._computedWatchers = {};
    Object.keys(computed).forEach(key => {
      Object.defineProperty(this._vm, key, {
        get: () => this._vm[key],
        enumerable: true
      });
    });
    this._vm.$options.computed = computed;
  }
  
  // 获取 state
  get state() {
    return this._vm._data.$$state;
  }
  
  // 设置 state (不允许直接修改)
  set state(v) {
    throw new Error('不能直接修改 state，请使用 mutations');
  }
  
  // commit 方法：执行 mutation
  commit(type, payload) {
    const entry = this._mutations[type];
    if (!entry) {
      console.error(`[Vuex] 未知的 mutation 类型: ${type}`);
      return;
    }
    
    // 执行 mutation 函数
    entry(this.state, payload);
  }
  
  // dispatch 方法：执行 action
  dispatch(type, payload) {
    const entry = this._actions[type];
    if (!entry) {
      console.error(`[Vuex] 未知的 action 类型: ${type}`);
      return;
    }
    
    // 执行 action 函数
    return entry(this, payload);
  }
}

// Vue 插件安装方法
Store.install = function(Vue) {
  Vue.mixin({
    beforeCreate() {
      const options = this.$options;
      if (options.store) {
        // 根组件
        this.$store = options.store;
      } else if (options.parent && options.parent.$store) {
        // 子组件
        this.$store = options.parent.$store;
      }
    }
  });
};

核心实现解析：

响应式状态：使用 Vue 实例的 data 属性存储 state，利用 Vue 的响应式系统实现状态变化时的自动更新。

Mutations：通过 commit 方法触发，是同步修改 state 的唯一方式，确保状态变更是可追踪的。

Actions：通过 dispatch 方法触发，可以包含异步操作，最终通过 commit 提交 mutation 来修改状态。

Getters：使用 Vue 的计算属性实现，具有缓存特性，只有当依赖的状态变化时才会重新计算。

插件系统：通过 Vue.mixin 在 beforeCreate 生命周期钩子中为所有组件注入 $store 属性。
